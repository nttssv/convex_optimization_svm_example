% ---------- Roadmap ----------
\begin{frame}{How the analysis is organized}
  \vspace{0.5em}
  \begin{center}
    \large
    \textcolor{UPAblue}{\textbf{hard margin}}
    $\,\rightarrow\,$
    \textcolor{BPAorange}{\textbf{check feasibility}}
    $\,\rightarrow\,$
    \textcolor{UPAblue}{\textbf{soft margin}}
    $\,\rightarrow\,$
    \textcolor{BPAorange}{\textbf{RBF comparison}}
    $\,\rightarrow\,$
    \textcolor{UPAblue}{\textbf{model choice}}
  \end{center}
  \vspace{1em}
  \begin{itemize}
    \item Start with a \textbf{hard-margin SVM}, which allows no classification errors.
    \item Check whether its constraints are feasible in three different ways.
    \item Move to a \textbf{soft-margin SVM} once the hard-margin model is ruled out.
    \item Compare the linear model with an \textbf{RBF-kernel SVM}.
    \item Choose the model using nested cross-validation.
  \end{itemize}
\end{frame}

% ---------- Steps at a glance ----------
\begin{frame}{Analysis plan}
  \small
  \begin{itemize}\setlength\itemsep{5pt}
    \item[\textcolor{UPAblue}{\textbf{1}}] \textbf{Test the hard-margin model.}
      It requires a linear boundary that classifies every observation correctly with a margin.
      We first ask whether those constraints are feasible.
    \item[\textcolor{BPAorange}{\textbf{2}}] \textbf{Confirm the infeasibility result.}
      Three checks show that the classes are not linearly separable:
      \emph{(2a)} the LP solver returns ``infeasible''; \emph{(2b)} a $12$-patient Farkas
      certificate turns that into a proof ($0\ge 1$); \emph{(2c)} the two convex hulls overlap.
    \item[\textcolor{UPAblue}{\textbf{3}}] \textbf{Fit a soft-margin model.}
      Slack variables allow margin violations, controlled by the penalty $C$.
      We tune $C$, inspect the fitted boundary, and also solve the primal with stochastic subgradients.
    \item[\textcolor{BPAorange}{\textbf{4}}] \textbf{Compare with an RBF kernel.}
      The kernel allows a nonlinear boundary. Nested cross-validation tests whether that
      added flexibility improves performance.
  \end{itemize}
  \take{The hard-margin analysis is a feasibility check; model performance is assessed only after moving to soft margins.}
\end{frame}

% ---------- Hard margin geometry ----------
\begin{frame}{Can a hard-margin SVM fit these data?}
  \lecref{Statistical Learning Models (W9); Convex Optimization Models (W8)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      The hard-margin SVM maximizes the distance between the two supporting hyperplanes:
      \[
        \min_{w,b}\ \tfrac12\lVert w\rVert^2
        \quad\text{s.t.}\quad t_i(w^\top x_i+b)\ge 1.
      \]
      \begin{itemize}
        \item Score $s=w^\top x+b$; sign is the prediction.
        \item Margin width $=2/\lVert w\rVert$.
        \item \textbf{Zero mistakes allowed.}
      \end{itemize}
      This is a convex quadratic program, provided the constraints are feasible.
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\textwidth]{output/figures/svm_primal_dual_toy_example.pdf}\\
      {\footnotesize Five-point teaching example: boundary, margins, support vectors.}
    \end{column}
  \end{columns}
\end{frame}

% ---------- Infeasibility ----------
\begin{frame}{Three checks for linear separability}
  \lecref{Introduction to Duality (W9); First-Order Optimality (W10)}
  \vspace{0.4em}
  We check the hard-margin constraints in \alert{three different ways}. All three reach the same conclusion.
  \vspace{0.8em}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{2a. Ask the solver}
        \small Run the feasibility LP on all data and every fold.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{2b. Farkas certificate}
        \small Use 12 nonzero weights to derive the contradiction $0\ge1$.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{2c. Convex hulls}
        \small Show that the two class hulls share a point.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.9em}
  \begin{center}
    \small The solver gives the numerical result, while the certificate and convex hulls verify it algebraically and geometrically.
  \end{center}
\end{frame}

% ---------- 2a solver ----------
\begin{frame}{The solver finds no feasible hard margin}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Feasibility of the hard-margin constraints is itself a \textbf{linear program}: does any $(w,b)$ satisfy
      \[
        t_i(w^\top x_i + b)\ge 1 \quad\text{for all } i?
      \]
      \begin{itemize}
        \item HiGHS returns \alert{\textbf{status: infeasible}} on the full 114-patient dataset.
        \item Repeated inside every one of the 5 cross-validation folds --- \textbf{all infeasible}.
        \item So the result is not an artifact of one particular split.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{block}{Solver verdict}
        \footnotesize
        \begin{tabular}{ll}
          \toprule
          Subset & Feasible? \\
          \midrule
          Full data (114) & \textcolor{BPAorange}{\textbf{No}} \\
          Fold 1 & \textcolor{BPAorange}{\textbf{No}} \\
          Fold 2 & \textcolor{BPAorange}{\textbf{No}} \\
          Fold 3 & \textcolor{BPAorange}{\textbf{No}} \\
          Fold 4 & \textcolor{BPAorange}{\textbf{No}} \\
          Fold 5 & \textcolor{BPAorange}{\textbf{No}} \\
          \bottomrule
        \end{tabular}
      \end{block}
    \end{column}
  \end{columns}
  \take{The hard-margin constraints are infeasible on the full dataset and in every training fold.}
\end{frame}

% ---------- 2b Farkas ----------
\begin{frame}{A Farkas certificate confirms infeasibility}
  \lecref{Introduction to Duality (W9) --- theorem of the alternative}
  \small
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \textbf{Farkas' lemma}: exactly \emph{one} of two systems can hold ---
      either a separating $w$ exists, \textbf{or} a set of nonnegative weights $\lambda$ gives a contradiction.
      \vspace{0.2em}
      \begin{itemize}
        \item The certificate has nonzero weights for \textbf{12 observations}, with $\lambda_i\ge0$.
        \item Combining their constraints gives $0\ge 1$, which is impossible.
        \item The calculation can be checked independently of the solver status.
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=0.94\textwidth]{output/figures/farkas_lemma_schematic.pdf}\\
      {\footnotesize Exactly one system can be feasible. For these data, it is the certificate system on the right.}
    \end{column}
  \end{columns}
  \take{The certificate proves that the hard-margin constraints are inconsistent.}
\end{frame}

% ---------- 2c convex hull ----------
\begin{frame}{The convex hulls explain the infeasibility}
  \small
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      After rescaling the same $\lambda$ weights, we obtain \textbf{the same point from both class convex hulls}:
      \[
        \textstyle\sum_i \beta_i^{+} x_i = \sum_i \beta_i^{-} x_i,
      \]
      matching to $<10^{-15}$ (machine precision).
      \begin{itemize}
        \item A shared point $\Rightarrow$ no hyperplane separates the classes.
        \item This gives a geometric interpretation of the solver and Farkas results.
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.52\textheight,keepaspectratio]{output/figures/convex_hull_overlap_2d.pdf}\\
      {\footnotesize The synthetic cohort in a 2-D projection. The exact certificate is computed in the full feature space.}
    \end{column}
  \end{columns}
\end{frame}

% ---------- Soft margin ----------
\begin{frame}{Soft margins allow violations}
  \lecref{Statistical Learning Models (W9) --- hinge loss, slack, penalty $C$}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      Slack variables $\xi_i$ allow observations to fall inside the margin or be misclassified:
      \[
        \min_{w,b,\xi}\ \tfrac12\lVert w\rVert^2 + C\!\sum_i \xi_i
      \]
      \[
        \text{s.t.}\ \ t_i(w^\top x_i+b)\ge 1-\xi_i,\ \ \xi_i\ge0.
      \]
      \begin{itemize}
        \item $C$ controls the penalty assigned to margin violations.
        \item Nested cross-validation selected \textbf{$C=0.01$}.
        \item The model can now accommodate overlap between the classes.
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\textwidth]{output/figures/linear_soft_margin_grid_roc.pdf}\\
      {\footnotesize (a) validation AUROC vs $C$; (b) nested out-of-fold ROC,
      seed-zero AUROC 0.95; 20-seed mean 0.94.}
    \end{column}
  \end{columns}
\end{frame}

% ---------- From constrained soft margin to stochastic subgradients ----------
\begin{frame}{From the constrained SVM to a subgradient objective}
  \lecref{Statistical Learning Models (W9); The Stochastic Gradient Method (W10)}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{1. Start with the constrained soft-margin problem}
      \[
        \min_{w,b,\xi}\ \tfrac12\lVert w\rVert^2+C\sum_i\xi_i,
        \quad
        \begin{aligned}
          t_i(w^\top x_i+b)&\ge 1-\xi_i,\\[-2pt]
          \xi_i&\ge0.
        \end{aligned}
      \]
      The smallest feasible slack is
      \[
        \xi_i=\max\!\left(0,1-t_i(w^\top x_i+b)\right),
      \]
      which is the \textbf{hinge loss} $\ell_i(w,b)$.
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{2. Substitute the hinge loss and rescale}
      \[
        \tfrac12\lVert w\rVert^2+C\sum_i\ell_i(w,b).
      \]
      Dividing by the positive constant $mC$ does not change the minimizer:
      \[
        P(w,b)=\frac1m\sum_i\ell_i(w,b)
        +\frac{\lambda}{2}\lVert w\rVert^2,
        \quad \lambda=\frac1{mC}.
      \]
      A stochastic method can therefore sample one observation and use its subgradient at each step.
    \end{column}
  \end{columns}
  \take{Because hinge loss is nondifferentiable at margin one, the Pegasos update is a stochastic subgradient step.}
\end{frame}

% ---------- RBF derivation ----------
\begin{frame}{An RBF kernel allows a nonlinear boundary}
  \lecref{Introduction to Duality (W9); Statistical Learning Models (W9) --- kernels}
  \small
  \textbf{1. Start from the soft-margin primal:}
  \[
    \min_{w,b,\xi}\ \tfrac12\lVert w\rVert^2 + C\sum_i \xi_i
    \quad\text{s.t.}\quad t_i(w^\top x_i+b)\ge 1-\xi_i,\ \ \xi_i\ge 0.
  \]
  \textbf{2. In the Lagrangian dual,} the data enter \alert{only through inner products} $x_i^\top x_j$:
  \[
    \max_{\alpha}\ \sum_i \alpha_i - \tfrac12\sum_{i,j}\alpha_i\alpha_j\, t_i t_j\,\underbrace{x_i^\top x_j}_{\text{inner product}}
    \quad\text{s.t.}\quad 0\le\alpha_i\le C,\ \ \sum_i\alpha_i t_i = 0.
  \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{3. Replace the inner product.} Map $x\!\mapsto\!\varphi(x)$ and use a \textbf{kernel}:
      \[
        x_i^\top x_j \;\longrightarrow\; K(x_i,x_j)=\varphi(x_i)^\top\varphi(x_j).
      \]
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{4. For the RBF kernel:}
      \[
        K(x_i,x_j)=\exp\!\big(-\gamma\lVert x_i-x_j\rVert^2\big).
      \]
      \small This gives a nonlinear boundary in the original space while the dual remains a convex QP.
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{center}
    Decision score becomes \;$f(x)=\displaystyle\sum_i \alpha_i t_i\,K(x_i,x)+b$.
  \end{center}
\end{frame}

% ---------- RBF compare ----------
\begin{frame}{RBF tuning favors the smoothest boundary}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      The RBF kernel $K(x_i,x_j)=e^{-\gamma\lVert x_i-x_j\rVert^2}$ can bend the boundary.
      \vspace{0.2em}
      \begin{itemize}
        \item Nested cross-validation tested 81 $(C,\gamma)$ pairs.
        \item The selected pair was $C=0.01,\ \gamma=0.001$, the \emph{smoothest} prespecified setting.
        \item An extended-grid audit continued toward smaller $C$ and inverse bandwidth $\gamma$: neither is precisely identified.
        \item Large $\gamma$ values fit the training data perfectly but perform near chance on validation data.
      \end{itemize}
      \vspace{0.1em}
      The selected RBF boundary is therefore close to linear.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.50\textheight,keepaspectratio]{output/figures/rbf_kernel_grid_heatmaps.pdf}\\
      {\footnotesize Validation AUROC (left) vs train--validation gap (right). The star marks the selected smooth model; the top rows overfit.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What the RBF boundary looks like}

% ---------- Conclusion ----------
\begin{frame}{The linear soft-margin model is sufficient here}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Results}
        \begin{itemize}
          \item The hard-margin problem is \textbf{infeasible}.
          \item Linear soft-margin: \textbf{20-seed mean AUROC 0.94}.
          \item RBF kernel: 20-seed mean AUROC 0.94.
          \item L2-logistic baseline: 20-seed mean AUROC 0.94.
        \end{itemize}
        \vspace{0.3em}
        \textbf{The RBF kernel does not improve mean nested outer-fold performance.}
      \end{block}
      With similar discrimination, the linear model is easier to tune and inspect.
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Why it matters}
        \small
        \begin{itemize}
          \item Hard-margin infeasibility reflects overlap in the feature space.
          \item Better training fit alone is not a reason to choose a more complex model.
          \item Because the data are synthetic, the result demonstrates the method rather than clinical performance.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
